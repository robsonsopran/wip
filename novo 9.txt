import os
import re
import webbrowser
from bs4 import BeautifulSoup

# Definição global do caminho da pasta e dos arquivos
PASTA = "./"  # Defina o caminho correto
ARQUIVO1 = os.path.join(PASTA, "ARQUIVO1.htm")
ARQUIVO2 = os.path.join(PASTA, "ARQUIVO2.htm")
ARQUIVO3 = os.path.join(PASTA, "ARQUIVO3.htm")
ARQUIVO4 = os.path.join(PASTA, "ARQUIVO4.htm")

def extrair_dados(arquivo, colunas):
    with open(arquivo, "r", encoding="utf-8") as f:
        soup = BeautifulSoup(f, "html.parser")
        tabela = soup.find("table")  # Supondo que os dados estejam em uma tabela
        
        if not tabela:
            return []
        
        dados = []
        for linha in tabela.find_all("tr")[1:]:  # Ignorando cabeçalho
            celulas = linha.find_all("td")
            if len(celulas) >= len(colunas):
                linha_dict = {coluna: str(celulas[i]) for i, coluna in enumerate(colunas)}  # Mantém a formatação HTML
                linha_dict["Links"] = linha_dict.get("Links", "")  # Se não existir, será string vazia
                dados.append(linha_dict)
        
        return dados

def buscar_no_arquivo1(object_id, dados_arquivo1):
    for linha in dados_arquivo1:
        if BeautifulSoup(linha["Object Identifier"], "html.parser").get_text(strip=True) == object_id:
            print(f"Encontrado no ARQUIVO1: {linha}")
            return linha
    return None

def buscar_nos_outros_arquivos(object_id, dados_arquivo, coluna_id):
    for linha in dados_arquivo:
        if BeautifulSoup(linha.get(coluna_id, ""), "html.parser").get_text(strip=True) == object_id:
            return linha
    return None

def extrair_links_validos(links, nome_arquivo):
    padrao = re.compile(rf"{nome_arquivo}-\\d+")
    return padrao.findall(links)

def gerar_html(resultados):
    html_path = os.path.join(PASTA, "resultado.html")
    with open(html_path, "w", encoding="utf-8") as f:
        f.write("<html><head><title>Resultados</title></head><body>")
        f.write("<h1>Resultados da Pesquisa</h1>")
        
        for nome_arquivo, dados in resultados.items():
            f.write(f"<h2>{nome_arquivo}</h2>")
            if dados:
                f.write("<table border='1'><tr>")
                f.write("".join(f"<th>{col}</th>" for col in dados.keys() if col != "Links"))
                f.write("</tr><tr>")
                f.write("".join(f"{val}" for col, val in dados.items() if col != "Links"))  # Mantendo formatação
                f.write("</tr></table>")
            else:
                f.write("<p>Object Identifier não encontrado.</p>")
        
        f.write("</body></html>")
    
    webbrowser.open(html_path)

def main():
    # Carregar os dados uma única vez
    colunas_arquivo1 = ["Object Identifier", "Statement", "Object Type", "Image Allocation", "Lane", "Links"]
    colunas_arquivo2 = ["Object Identifier", "Statement", "Image Allocation"]
    colunas_arquivo3 = ["ID", "Platsr"]
    colunas_arquivo4 = ["ID", "ICD-SW", "Detailed Description", "Allocation", "Acess", "Adress"]
    
    print("Carregando arquivos...")
    dados_arquivo1 = extrair_dados(ARQUIVO1, colunas_arquivo1)
    dados_arquivo2 = extrair_dados(ARQUIVO2, colunas_arquivo2)
    dados_arquivo3 = extrair_dados(ARQUIVO3, colunas_arquivo3)
    dados_arquivo4 = extrair_dados(ARQUIVO4, colunas_arquivo4)
    
    while True:
        object_id = input("Digite o Object Identifier para pesquisa (ou 'sair' para encerrar): ")
        if object_id.lower() == "sair":
            break
        
        resultado_arquivo1 = buscar_no_arquivo1(object_id, dados_arquivo1)
        
        if not resultado_arquivo1:
            print("Object Identifier não encontrado no ARQUIVO1.")
            continue
        
        print("\nResultados encontrados no ARQUIVO1:")
        print(resultado_arquivo1)
        
        # Extrair links válidos
        links = resultado_arquivo1.get("Links", "")
        arquivos_referenciados = {
            "ARQUIVO2": extrair_links_validos(links, "ARQUIVO2"),
            "ARQUIVO3": extrair_links_validos(links, "ARQUIVO3"),
            "ARQUIVO4": extrair_links_validos(links, "ARQUIVO4"),
        }
        
        resultados_finais = {"ARQUIVO1": {key: val for key, val in resultado_arquivo1.items() if key != "Links"}}
        
        resultados_finais["ARQUIVO2"] = buscar_nos_outros_arquivos(object_id, dados_arquivo2, "Object Identifier")
        resultados_finais["ARQUIVO3"] = buscar_nos_outros_arquivos(object_id, dados_arquivo3, "ID")
        resultados_finais["ARQUIVO4"] = buscar_nos_outros_arquivos(object_id, dados_arquivo4, "ID")
        
        gerar_html(resultados_finais)

if __name__ == "__main__":
    main()
